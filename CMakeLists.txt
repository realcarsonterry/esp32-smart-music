cmake_minimum_required(VERSION 3.20)
project(smart_music_core VERSION 1.0 LANGUAGES C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)

add_library(music_core STATIC core/music_core.c)
target_include_directories(music_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/core)
# Apply strict floating point settings after any caller-supplied global flags.
if(MSVC)
    target_compile_options(music_core PUBLIC /fp:strict PRIVATE /W4 /WX)
else()
    target_compile_options(music_core PUBLIC -fno-fast-math
        PRIVATE -Wall -Wextra -Wpedantic -Werror)
    target_link_libraries(music_core PUBLIC m)
endif()

add_library(simulator_cjson STATIC simulator/vendor/cjson/cJSON.c)
target_include_directories(simulator_cjson PUBLIC simulator/vendor/cjson)
target_compile_definitions(simulator_cjson PRIVATE CJSON_HIDE_SYMBOLS)
add_library(simulator_json STATIC simulator/music_json.c)
target_include_directories(simulator_json PUBLIC simulator)
target_link_libraries(simulator_json PUBLIC music_core PRIVATE simulator_cjson)
add_executable(simulator simulator/main.c)
target_link_libraries(simulator PRIVATE simulator_json)
if(MSVC)
    target_compile_definitions(simulator_json PRIVATE _CRT_SECURE_NO_WARNINGS)
    target_compile_definitions(simulator PRIVATE _CRT_SECURE_NO_WARNINGS)
    target_compile_options(simulator_json PRIVATE /W4 /WX)
    target_compile_options(simulator PRIVATE /W4 /WX)
else()
    target_compile_options(simulator_json PRIVATE -Wall -Wextra -Wpedantic -Werror)
    target_compile_options(simulator PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(core_tests tests/core_tests.c)
    target_link_libraries(core_tests PRIVATE music_core)
    if(MSVC)
        target_compile_options(core_tests PRIVATE /W4 /WX)
    else()
        target_compile_options(core_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
    foreach(case IN ITEMS ticks rng valid bpm midi time polyphony protocol
            structure chords context pcm wav)
        add_test(NAME core_${case} COMMAND core_tests ${case})
    endforeach()
    add_executable(ks_tests tests/ks_tests.c)
    target_link_libraries(ks_tests PRIVATE music_core)
    if(MSVC)
        target_compile_options(ks_tests PRIVATE /W4 /WX)
    else()
        target_compile_options(ks_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
    foreach(case IN ITEMS pitch velocity release limits blocks seed_reset)
        add_test(NAME ks_${case} COMMAND ks_tests ${case})
    endforeach()
    add_test(NAME simulator_cli COMMAND ${CMAKE_COMMAND}
        -DSIMULATOR=$<TARGET_FILE:simulator>
        -DSOURCE_DIR=${CMAKE_CURRENT_SOURCE_DIR}
        -DTEST_DIR=${CMAKE_CURRENT_BINARY_DIR}/simulator-tests
        -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/simulator_tests.cmake)
endif()
